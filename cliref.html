Recursively list files only by date: find . -type f -exec ls -lt {} +
Recursively list files only by date (nothing hidden): find . -path '*/.*' -prune -o -type f -print0 | xargs -0 ls -lt
Find named folders: find . -type d -name \*xyz\*
Recursively delete named folders: find . -type d -depth -name .svn -exec rm -rf {} \;
